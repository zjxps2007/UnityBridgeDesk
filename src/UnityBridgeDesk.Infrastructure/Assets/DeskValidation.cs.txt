using System;
using System.Collections.Generic;
using System.Linq;
using System.Threading.Tasks;
using Newtonsoft.Json.Linq;
using UnityEditor;
using UnityEditor.SceneManagement;
using UnityEngine;
using UnityBridgeConnector;

[UnityBridgeTool(Name = "desk_validate", Description = "Read-only observation and public button input for Desk draft-v1 tasks")]
public static class DeskValidation
{
    public static async Task<object> HandleCommand(JObject p)
    {
        var errors = new List<string>();
        string task = (string)p["task"], mode = (string)p["mode"];
        int stage = (int?)p["stage"] ?? 1;
        try
        {
            if (mode == "reopen")
            {
                if (EditorApplication.isPlaying) throw new Exception("Must be in Edit mode");
                if (!System.IO.File.Exists("Assets/DeskTask.unity")) throw new Exception("Saved Assets/DeskTask.unity missing");
                // Reopen without saving: validation never fixes or saves the submission.
                EditorSceneManager.OpenScene("Assets/DeskTask.unity", OpenSceneMode.Single);
                if (task == "A01") InspectDrop(errors);
                else if (task == "A02") InspectPrefab(errors, stage);
                else InspectMachine(errors, stage);
            }
            else if (mode == "exercise" && task == "A01")
            {
                if (!EditorApplication.isPlaying) throw new Exception("Play mode required");
                var cube = Find("Cube"); var floor = Find("Floor").GetComponent<Collider>();
                float start = DeskPassive.StartHeight;
                await Task.Delay(3000);
                var collider = cube.GetComponent<Collider>(); var body = cube.GetComponent<Rigidbody>();
                if (!DeskPassive.Observed || !(cube.transform.position.y < start - .1f)) errors.Add("Cube did not fall from observed position");
                if (Mathf.Abs(collider.bounds.min.y - floor.bounds.max.y) > .15f || body.linearVelocity.magnitude > .2f) errors.Add("Cube did not settle on floor");
            }
            else if (mode == "exercise" && task == "A03") await ExerciseMachine(errors, stage);
            else errors.Add("Unsupported observation request");
        }
        catch (Exception e) { errors.Add(e.GetBaseException().Message); }
        return new SuccessResponse("Independent validation", new { nonce = (string)p["nonce"],
            projectPath = System.IO.Path.GetDirectoryName(Application.dataPath), pid = System.Diagnostics.Process.GetCurrentProcess().Id,
            passed = errors.Count == 0, errors = errors.ToArray() });
    }
    static GameObject Find(string name) => GameObject.Find(name) ?? throw new Exception("Missing active object: " + name);
    static Rigidbody[] Bodies() => Enumerable.Range(0,16).Select(i => Find("Cube_" + i.ToString("D2")).GetComponent<Rigidbody>() ?? throw new Exception("Rigidbody missing")).ToArray();
    static void InspectDrop(List<string> errors)
    {
        var floor = Find("Floor"); var cube = Find("Cube");
        var body = cube.GetComponent<Rigidbody>();
        if (floor.GetComponent<Collider>() == null || cube.GetComponent<Collider>() == null || body == null || !body.useGravity || body.isKinematic) errors.Add("Collider or dynamic gravity missing");
        if (Vector3.Distance(cube.transform.position, new Vector3(0,3,0)) > .01f || Vector3.Distance(cube.transform.localScale,Vector3.one) > .01f) errors.Add("Initial cube pose/size mismatch");
        if (Mathf.Abs(floor.transform.position.y) > .01f || floor.GetComponent<Collider>().isTrigger) errors.Add("Floor pose/solid collider mismatch");
        if (UnityEngine.Object.FindObjectsByType<Camera>(FindObjectsSortMode.None).Length == 0) errors.Add("Camera missing");
    }
    static void InspectPrefab(List<string> errors, int stage)
    {
        var objects = UnityEngine.Object.FindObjectsByType<Transform>(FindObjectsSortMode.None).Where(x => x.name.StartsWith("Item_")).ToArray();
        if (objects.Length != 100) errors.Add("Exactly 100 item IDs required");
        string prefab = null;
        for (int i=0;i<100;i++)
        {
            var go = Find("Item_"+i.ToString("D3")); bool exception = stage >= 2 && i < 20;
            if (!PrefabUtility.IsPartOfPrefabInstance(go)) { errors.Add("Prefab connection missing: "+i); continue; }
            string path = PrefabUtility.GetPrefabAssetPathOfNearestInstanceRoot(go);
            if (prefab == null) prefab = path; else if (prefab != path) errors.Add("Different base prefab: "+i);
            var expected = new Vector3((i%10)*2, exception ? 1 : 0, (i/10)*2);
            if (Vector3.Distance(go.transform.position,expected) > .01f) errors.Add("Grid/height mismatch: "+i);
            var renderer = go.GetComponent<Renderer>();
            Color color = renderer.sharedMaterial.color;
            if (Vector4.Distance(color, exception ? Color.red : Color.white) > .02f) errors.Add("Color exception mismatch: "+i);
            var box = go.GetComponent<BoxCollider>();
            if (box == null || Vector3.Distance(box.size,stage >=3 ? new Vector3(2,1,1) : Vector3.one) > .01f) errors.Add("Shared collider mismatch: "+i);
        }
        if (prefab != null && stage >=3)
        {
            var asset = AssetDatabase.LoadAssetAtPath<GameObject>(prefab);
            if (Vector3.Distance(asset.GetComponent<BoxCollider>().size,new Vector3(2,1,1)) > .01f) errors.Add("Base prefab collider not changed");
        }
    }
    static void InspectMachine(List<string> errors, int stage)
    {
        if (UnityEngine.Object.FindObjectsByType<Rigidbody>(FindObjectsSortMode.None).Length != 16) errors.Add("Exactly 16 dynamic cubes required");
        var bodies = Bodies();
        for (int i=0;i<16;i++)
        {
            if (Vector3.Distance(bodies[i].position,Initial(i)) > .01f || bodies[i].GetComponent<Collider>() == null) errors.Add("Initial cube mismatch: "+i);
        }
        Find("Floor"); Find("Count"); Find("Start"); Find("Reset");
        if(!UnityEngine.Object.FindObjectsByType<Component>(FindObjectsSortMode.None).Any(x=>x!=null && x.GetType().FullName=="UnityEngine.EventSystems.EventSystem"))errors.Add("EventSystem missing");
        if (stage >=2) { Find("Pause"); Find("Resume"); }
    }
    static Vector3 Initial(int i) => new Vector3((i%4)*2, 6, (i/4)*2);
    static void Click(string name)
    {
        var go = Find(name);
        var component = go.GetComponents<Component>().FirstOrDefault(x => x != null && x.GetType().FullName == "UnityEngine.UI.Button");
        if (component == null) throw new Exception("Public uGUI Button missing: "+name);
        if(component is Behaviour behaviour && !behaviour.isActiveAndEnabled)throw new Exception("Button disabled: "+name);
        var type = component.GetType();
        if (!(bool)type.GetProperty("interactable").GetValue(component)) throw new Exception("Button not interactable: "+name);
        object click = type.GetProperty("onClick").GetValue(component);
        click.GetType().GetMethod("Invoke").Invoke(click,null);
    }
    static int Count()
    {
        var component = Find("Count").GetComponents<Component>().FirstOrDefault(x => x != null && x.GetType().GetProperty("text")?.PropertyType == typeof(string));
        if (component == null || !int.TryParse((string)component.GetType().GetProperty("text").GetValue(component),out int count)) throw new Exception("Count must be a visible integer text");
        if(component is Behaviour behaviour && !behaviour.isActiveAndEnabled)throw new Exception("Count text disabled");
        if(component.GetComponentInParent<Canvas>()==null)throw new Exception("Count text has no Canvas");
        var font=component.GetType().GetProperty("font");if(font!=null && font.GetValue(component)==null)throw new Exception("Count font missing");
        return count;
    }
    static async Task ExerciseMachine(List<string> errors, int stage)
    {
        if (!EditorApplication.isPlaying) throw new Exception("Play mode required");
        Click("Reset");
        for (int cycle=0;cycle<2;cycle++)
        {
            CheckReset(errors);
            if(stage>=2){Click("Pause");await Task.Delay(200);CheckReset(errors);Click("Resume");}
            await Task.Delay(350);CheckReset(errors);
            Click("Start");
            if (stage >=2)
            {
                await Task.Delay(150);
                Click("Pause");
                var positions = Bodies().Select(x=>x.position).ToArray(); int before = Count();
                if (before >=16) errors.Add("Pause was not observed before landing");
                await Task.Delay(800);
                if (Bodies().Where((x,i)=>Vector3.Distance(x.position,positions[i])>.01f).Any() || Count()!=before) errors.Add("Paused state advanced");
                Click("Reset"); await Task.Delay(350);CheckReset(errors); // Reset must stay idle, including from pause.
                Click("Start"); await Task.Delay(150); Click("Pause"); Click("Resume");
            }
            await Task.Delay(4000);
            float ground = Find("Floor").GetComponent<Collider>().bounds.max.y;
            int landed = Bodies().Count(x=>Mathf.Abs(x.GetComponent<Collider>().bounds.min.y-ground)<.15f && x.linearVelocity.magnitude<.2f);
            if (landed !=16 || Count()!=landed) errors.Add("Contact count disagrees with observed cubes: "+cycle);
            await Task.Delay(1000);
            if (Count()!=16) errors.Add("Repeated contacts were counted twice");
            Click("Reset"); CheckReset(errors);
        }
    }
    static void CheckReset(List<string> errors)
    {
        if (Count()!=0) errors.Add("Reset count not zero");
        var bodies = Bodies();
        for (int i=0;i<16;i++) if (Vector3.Distance(bodies[i].position,Initial(i))>.01f || bodies[i].linearVelocity.magnitude>.01f || bodies[i].angularVelocity.magnitude>.01f) errors.Add("Reset pose/velocity mismatch: "+i);
    }
}

[InitializeOnLoad]
public static class DeskPassive
{
    public static float StartHeight;
    public static bool Observed;
    static DeskPassive()
    {
        EditorApplication.playModeStateChanged += state =>
        {
            if (state == PlayModeStateChange.EnteredPlayMode)
            {
                var cube = GameObject.Find("Cube"); Observed = cube != null;
                if (cube != null) StartHeight = cube.transform.position.y;
            }
        };
    }
}
