using System;
using System.Linq;
using Newtonsoft.Json.Linq;
using UnityEditor;
using UnityEditor.SceneManagement;
using UnityEngine;
using UnityBridgeConnector;

[UnityBridgeTool(Name = "desk_probe", Description = "UnityBridge Desk fixed experiment fixture, draft-v1")]
public static class DeskProbe
{
    public static object HandleCommand(JObject p)
    {
        object value;
        string action = (string)p["action"];
        if (action == "echo") value = 42;
        else if (action == "payload")
        {
            int size = (int)p["size"];
            if (size < 0 || size > 1048576) throw new ArgumentException("size");
            value = new string(Enumerable.Range(0, size).Select(i => (char)('A' + i % 26)).ToArray());
        }
        else if (action == "create")
        {
            EditorSceneManager.NewScene(NewSceneSetup.EmptyScene, NewSceneMode.Single);
            int count = (int)p["count"];
            for (int i = 0; i < count; i++) { var go = new GameObject("DeskFixed_" + i); go.transform.position = new Vector3(0, -1, 0); }
            value = count;
        }
        else if (action == "move")
        {
            int start = (int)p["start"], count = (int)p["count"];
            for (int i = start; i < start + count; i++) GameObject.Find("DeskFixed_" + i).transform.position = Expected(i);
            value = count;
        }
        else if (action == "inspect")
        {
            int count = (int)p["count"];
            var objects = UnityEngine.SceneManagement.SceneManager.GetActiveScene().GetRootGameObjects().SelectMany(x=>x.GetComponentsInChildren<Transform>(true)).ToArray();
            value = objects.Length == count && Enumerable.Range(0, count).All(i =>
                objects.Count(x => x.name == "DeskFixed_" + i && Vector3.Distance(x.position, Expected(i)) < .00001f) == 1);
        }
        else if (action == "revision") value = DeskRevision.Value;
        else if (action == "stress_objects")
        {
            int count = (int)p["count"];
            if (count < 1 || count > 10000) throw new ArgumentException("count");
            var objects = new GameObject[count];
            try
            {
                for (int i = 0; i < count; i++)
                {
                    objects[i] = new GameObject("DeskStress_" + i);
                    objects[i].transform.position = Expected(i);
                }
                value = count;
            }
            finally { foreach (var obj in objects) if (obj != null) UnityEngine.Object.DestroyImmediate(obj); }
        }
        else if (action == "state") value = new { playing = EditorApplication.isPlaying, paused = EditorApplication.isPaused, compiling = EditorApplication.isCompiling };
        else return new ErrorResponse("Unknown fixture action");
        return new SuccessResponse("Desk fixture", new { nonce = (string)p["nonce"], projectPath = System.IO.Path.GetDirectoryName(Application.dataPath), pid = System.Diagnostics.Process.GetCurrentProcess().Id, value });
    }
    static Vector3 Expected(int i) => new Vector3(i % 100, i / 100, i % 7);
}
