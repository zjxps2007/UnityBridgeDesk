using System;
using System.Linq;
using Newtonsoft.Json.Linq;
using UnityEditor;
using UnityEditor.SceneManagement;
using UnityEngine;
#if DESK_PIPELINE
using Unity.Pipeline.Commands;
#elif DESK_GO_CLI
using UnityCliConnector;
#else
using UnityBridgeConnector;
#endif

#if DESK_GO_CLI
[UnityCliTool(Name = "desk_probe", Description = "UnityBridge Desk fixed experiment fixture, draft-v1")]
#elif !DESK_PIPELINE
[UnityBridgeTool(Name = "desk_probe", Description = "UnityBridge Desk fixed experiment fixture, draft-v1")]
#endif
public static class DeskProbe
{
#if DESK_PIPELINE
    [CliCommand("desk_probe", "Desk shared benchmark workload", MainThreadRequired = true)]
    public static object PipelineCommand([CliArg("parameters", "JSON workload input", Required = true)] string parameters)
        => Run(JObject.Parse(parameters));
#else
    public static object HandleCommand(JObject p)
        => new SuccessResponse("Desk fixture", Run(p));
#endif
    public static object Run(JObject p)
    {
        object value;
        string action = (string)p["action"];
        double delayElapsedMs = 0;
        if (action == "echo")
        {
            int delay = p["delayMs"] == null ? 0 : (int)p["delayMs"];
            if (delay != 0 && delay != 100) throw new ArgumentException("delayMs");
            if (delay > 0)
            {
                var timer = System.Diagnostics.Stopwatch.StartNew();
                System.Threading.Thread.Sleep(delay);
                delayElapsedMs = timer.Elapsed.TotalMilliseconds;
            }
            value = 42;
        }
        else if (action == "payload")
        {
            int size = (int)p["size"];
            if (size < 0 || size > 1048576) throw new ArgumentException("size");
            value = new string(Enumerable.Range(0, size).Select(i => (char)('A' + i % 26)).ToArray());
        }
        else if (action == "create")
        {
            EditorSceneManager.NewScene(NewSceneSetup.EmptyScene, NewSceneMode.Single);
            int count = (int)p["count"];
            for (int i = 0; i < count; i++) { var go = new GameObject("DeskFixed_" + i); go.transform.position = new Vector3(0, -1, 0); }
            value = count;
        }
        else if (action == "move")
        {
            int start = (int)p["start"], count = (int)p["count"];
            for (int i = start; i < start + count; i++) GameObject.Find("DeskFixed_" + i).transform.position = Expected(i);
            value = count;
        }
        else if (action == "inspect")
        {
            int count = (int)p["count"];
            var objects = UnityEngine.SceneManagement.SceneManager.GetActiveScene().GetRootGameObjects().SelectMany(x=>x.GetComponentsInChildren<Transform>(true)).ToArray();
            value = objects.Length == count && Enumerable.Range(0, count).All(i =>
                objects.Count(x => x.name == "DeskFixed_" + i && Vector3.Distance(x.position, Expected(i)) < .00001f) == 1);
        }
        else if (action == "snapshot")
        {
            value = UnityEngine.SceneManagement.SceneManager.GetActiveScene().GetRootGameObjects()
                .SelectMany(x => x.GetComponentsInChildren<Transform>(true))
                .Select(x => new { name = x.name, x = x.position.x, y = x.position.y, z = x.position.z }).ToArray();
        }
        else if (action == "revision") value = DeskRevision.Value;
        else if (action == "stress_objects")
        {
            int count = (int)p["count"];
            if (count < 1 || count > 10000) throw new ArgumentException("count");
            var objects = new GameObject[count];
            try
            {
                for (int i = 0; i < count; i++)
                {
                    objects[i] = new GameObject("DeskStress_" + i);
                    objects[i].transform.position = Expected(i);
                }
                value = count;
            }
            finally { foreach (var obj in objects) if (obj != null) UnityEngine.Object.DestroyImmediate(obj); }
        }
        else if (action == "state") value = new { playing = EditorApplication.isPlaying, paused = EditorApplication.isPaused, compiling = EditorApplication.isCompiling };
        else throw new ArgumentException("Unknown fixture action");
        return new { nonce = (string)p["nonce"], projectPath = System.IO.Path.GetDirectoryName(Application.dataPath), pid = System.Diagnostics.Process.GetCurrentProcess().Id, value, delayElapsedMs };
    }
    static Vector3 Expected(int i) => new Vector3(i % 100, i / 100, i % 7);
}

#if DESK_PIPELINE
// Observe editor readiness without invoking a benchmark command. Never persist the server token.
[InitializeOnLoad]
static class DeskPipelineReadiness
{
    static double next;
    static DeskPipelineReadiness() { EditorApplication.update += Tick; }
    static void Tick()
    {
        if (EditorApplication.timeSinceStartup < next) return;
        next = EditorApplication.timeSinceStartup + .5;
        var package = UnityEditor.PackageManager.PackageInfo.FindForAssembly(typeof(CliCommandAttribute).Assembly);
        var project = System.IO.Path.GetDirectoryName(Application.dataPath);
        var data = new { projectPath = project, pid = System.Diagnostics.Process.GetCurrentProcess().Id,
            timestamp = DateTimeOffset.UtcNow.ToUnixTimeMilliseconds(), unityVersion = Application.unityVersion,
            connectorVersion = package?.version, compileErrors = EditorUtility.scriptCompilationFailed,
            state = EditorApplication.isCompiling || EditorApplication.isUpdating ? "compiling" :
                EditorApplication.isPaused ? "paused" : EditorApplication.isPlaying ? "playing" : "ready" };
        try { System.IO.File.WriteAllText(System.IO.Path.Combine(project, "Library", "desk-pipeline-ready.json"), Newtonsoft.Json.JsonConvert.SerializeObject(data)); }
        catch (System.IO.IOException) { }
    }
}
#endif
